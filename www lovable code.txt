      el.style.height = (el.scrollHeight) + 'px';
    }

    function insertPrompt(promptText) {
      chatTextarea.value = promptText;
      adjustTextareaHeight(chatTextarea);
      chatTextarea.focus();
    }

    function initSessionHistory() {
      const storedHistory = localStorage.getItem('harshal_session_history_v2');
      if (storedHistory) {
        activeChatHistory = JSON.parse(storedHistory);
      } else {
        activeChatHistory = [
          {
            id: 'sh-1',
            title: "Exploring A-Z Possibilities",
            timestamp: Date.now() - 3600000,
            messages: [
              { sender: 'user', text: 'How do we solve trash issues in my residential area?' },
              { sender: 'ai', text: 'To solve waste issues permanently, we use Chapter S: Security with Sustainability (protecting today\'s needs without compromising tomorrow) and decentralized upcycling. Feel free to explore other chapters!' }
            ]
          }
        ];
        saveHistoryToStorage();
      }
      renderHistoryList();
    }

    function saveHistoryToStorage() {
      localStorage.setItem('harshal_session_history_v2', JSON.stringify(activeChatHistory));
    }

    function renderHistoryList() {
      historyList.innerHTML = '';
      if (activeChatHistory.length === 0) {
        historyList.innerHTML = `
          <div class="text-[11px] text-zinc-600 px-2 italic">
            No recent sessions.
          </div>
        `;
        return;
      }

      activeChatHistory.forEach(session => {
        const item = document.createElement('button');
        item.className = "w-full text-left p-2 rounded-lg text-xs hover:bg-zinc-900 transition flex items-center justify-between text-zinc-400 hover:text-zinc-200 group";
        item.onclick = () => loadSession(session.id);
        item.innerHTML = `
          <div class="flex items-center gap-2 truncate">
            <svg class="w-3.5 h-3.5 text-zinc-600 shrink-0" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M8 12h.01M12 12h.01M16 12h.01M21 12c0 4.418-4.03 8-9 8a9.863 9.863 0 01-4.255-.949L3 20l1.395-3.72C3.512 15.042 3 13.574 3 12c0-4.418 4.03-8 9-8s9 3.582 9 8z" /></svg>
            <span class="truncate">${session.title}</span>
          </div>
          <button onclick="event.stopPropagation(); deleteSession('${session.id}');" class="opacity-0 group-hover:opacity-100 p-0.5 hover:text-red-400 transition" title="Delete session">
            <svg class="w-3 h-3" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12" /></svg>
          </button>
        `;
        historyList.appendChild(item);
      });
    }

    function loadSession(id) {
      const session = activeChatHistory.find(s => s.id === id);
      if (!session) return;
      
      welcomeScreen.classList.add('hidden');
      messagesContainer.classList.remove('hidden');
      messagesContainer.innerHTML = '';

      session.messages.forEach(msg => {
        appendMessage(msg.sender, msg.text);
      });

      showToast(`Loaded exploration: "${session.title}"`);
    }

    function deleteSession(id) {
      activeChatHistory = activeChatHistory.filter(s => s.id !== id);
      saveHistoryToStorage();
      renderHistoryList();
      
      welcomeScreen.classList.remove('hidden');
      messagesContainer.classList.add('hidden');
      messagesContainer.innerHTML = '';
      showToast('Session cleared from local vault.');
    }

    function clearChatHistory() {
      activeChatHistory = [];
      saveHistoryToStorage();
      renderHistoryList();
      welcomeScreen.classList.remove('hidden');
      messagesContainer.classList.add('hidden');
      messagesContainer.innerHTML = '';
      showToast('All cached sessions wiped successfully.');
    }

    function appendMessage(sender, text) {
      const isUser = sender === 'user';
      const msgWrapper = document.createElement('div');
      msgWrapper.className = `flex gap-4 ${isUser ? 'justify-end' : 'justify-start'}`;

      const avatar = isUser 
        ? `<div class="w-8 h-8 rounded-full border theme-border-color bg-zinc-900 flex items-center justify-center font-bold text-[10px] text-zinc-400 uppercase shrink-0">U</div>`
        : `<div class="w-8 h-8 rounded-lg theme-accent-bg flex items-center justify-center font-bold text-xs text-black shrink-0 font-display">H</div>`;

      const formattedText = text.replace(/\n/g, '<br>');

      const bubble = `
        <div class="flex items-start gap-3 max-w-xl ${isUser ? 'flex-row-reverse' : 'flex-row'}">
          ${avatar}
          <div class="space-y-1">
            <div class="${isUser ? 'bg-zinc-900 border theme-border-color text-zinc-100' : 'theme-panel-bg border theme-border-color text-zinc-200'} px-4 py-3 rounded-2xl text-xs leading-relaxed shadow-sm">
              <p>${formattedText}</p>
            </div>
            <div class="text-[9px] text-zinc-600 ${isUser ? 'text-right' : 'text-left'} font-mono px-1">
              ${new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})}
            </div>
          </div>
        </div>
      `;

      msgWrapper.innerHTML = bubble;
      messagesContainer.appendChild(msgWrapper);
      
      const scrollArea = document.getElementById('chat-messages');
      scrollArea.scrollTo({ top: scrollArea.scrollHeight, behavior: 'smooth' });
    }

    function removeTypingIndicator() {
      if (typingIndicatorElement) {
        typingIndicatorElement.remove();
        typingIndicatorElement = null;
      }
    }

    async function submitChat() {
      const userText = chatTextarea.value.trim();
      if (!userText) return;

      welcomeScreen.classList.add('hidden');
      messagesContainer.classList.remove('hidden');

      appendMessage('user', userText);
      chatTextarea.value = '';
      adjustTextareaHeight(chatTextarea);

      let currentSessionId = messagesContainer.dataset.sessionId;
      if (!currentSessionId) {
        currentSessionId = 'session_' + Date.now();
        messagesContainer.dataset.sessionId = currentSessionId;
        activeChatHistory.unshift({
          id: currentSessionId,
          title: userText.substring(0, 30) + (userText.length > 30 ? '...' : ''),
          timestamp: Date.now(),
          messages: []
        });
      }

      const activeSession = activeChatHistory.find(s => s.id === currentSessionId);
      if (activeSession) {
        activeSession.messages.push({ sender: 'user', text: userText });
      }

      showTypingIndicator();

      let aiResponseText = "";
      try {
        if (apiKey && apiKey !== "") {
          aiResponseText = await queryGeminiAPI(userText);
        } else {
          aiResponseText = await getHarSHalSimulatedResponse(userText);
        }
      } catch (err) {
        console.error("Query issue, invoking internal engine:", err);
        aiResponseText = await getHarSHalSimulatedResponse(userText);
      }

      removeTypingIndicator();
      appendMessage('ai', aiResponseText);

      if (activeSession) {
        activeSession.messages.push({ sender: 'ai', text: aiResponseText });
        saveHistoryToStorage();
        renderHistoryList();
      }
    }
